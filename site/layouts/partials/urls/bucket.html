Numbers arriving from JSON are float64, so cast before formatting. */ -}} {{- $n := int .number }} {{- $size := int .size }} {{- $lower := int (mul (math.Floor (div $n $size)) $size) }} {{- return (printf "%d-%d" $lower (add $lower (sub $size 1))) }}
